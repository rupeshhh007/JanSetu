import com.jansetu.backend.scheme.entity.EligibilityCriteria;
import org.junit.jupiter.api.BeforeEach;
import org.junit.jupiter.api.Test;

import java.math.BigDecimal;
import java.time.LocalDate;
import java.util.List;

import static org.junit.jupiter.api.Assertions.*;

class EligibilityEvaluatorTest {

    private EligibilityEvaluator evaluator;
    private CitizenProfile profile;

    @BeforeEach
    void setUp() {
        evaluator = new EligibilityEvaluator();
        profile = new CitizenProfile();

        // Default valid profile setup
        profile.setDateOfBirth(LocalDate.now().minusYears(20)); // 20 years old
        profile.setAnnualFamilyIncome(new BigDecimal("200000.00"));
        profile.setCategory(Category.SC);
        profile.setPrimaryOccupation(Occupation.STUDENT);
        profile.setStudent(true);
        profile.setFarmer(false);
        profile.setHasDisability(false);
    }

    @Test
    void shouldReturnTrueWhenAllCriteriaMatch() {
        EligibilityCriteria criteria = new EligibilityCriteria();
        criteria.setMinAge(18);
        criteria.setMaxFamilyIncome(new BigDecimal("250000.00"));
        criteria.setRequiredCategories(List.of(Category.SC, Category.ST));
        criteria.setRequiresStudent(true);

        assertTrue(evaluator.isEligible(profile, criteria));
    }

    @Test
    void shouldReturnFalseWhenIncomeIsTooHigh() {
        EligibilityCriteria criteria = new EligibilityCriteria();
        criteria.setMaxFamilyIncome(new BigDecimal("150000.00")); // Criteria is 1.5L, Profile is 2L

        assertFalse(evaluator.isEligible(profile, criteria));
    }

    @Test
    void shouldReturnFalseWhenAgeIsBelowMinimum() {
        EligibilityCriteria criteria = new EligibilityCriteria();
        criteria.setMinAge(25); // Criteria is 25, Profile is 20

        assertFalse(evaluator.isEligible(profile, criteria));
    }

    @Test
    void shouldReturnFalseWhenCategoryDoesNotMatch() {
        EligibilityCriteria criteria = new EligibilityCriteria();
        criteria.setRequiredCategories(List.of(Category.GENERAL)); // Profile is SC

        assertFalse(evaluator.isEligible(profile, criteria));
    }

    @Test
    void shouldReturnTrueWhenCriteriaAreNull() {
        // If criteria fields are null, they should be ignored (the rule is not enforced)
        EligibilityCriteria criteria = new EligibilityCriteria();

        assertTrue(evaluator.isEligible(profile, criteria));
    }
}

package com.jansetu.backend.scheme.evaluator;

import com.jansetu.backend.profile.entity.CitizenProfile;
import com.jansetu.backend.scheme.entity.EligibilityCriteria;
import org.springframework.stereotype.Component;

import java.time.LocalDate;
import java.time.Period;

@Component
public class EligibilityEvaluator {

    public boolean isEligible(CitizenProfile profile, EligibilityCriteria criteria) {

        // 1. Age Check
        if (criteria.getMinAge() != null || criteria.getMaxAge() != null) {
            int age = Period.between(profile.getDateOfBirth(), LocalDate.now()).getYears();

            if (criteria.getMinAge() != null && age < criteria.getMinAge()) return false;
            if (criteria.getMaxAge() != null && age > criteria.getMaxAge()) return false;
        }

        // 2. Income Check
        if (criteria.getMaxFamilyIncome() != null) {
            // compareTo returns -1 if less, 0 if equal, 1 if greater
            if (profile.getAnnualFamilyIncome().compareTo(criteria.getMaxFamilyIncome()) > 0) {
                return false;
            }
        }

        // 3. Category Check
        if (criteria.getRequiredCategories() != null && !criteria.getRequiredCategories().isEmpty()) {
            if (!criteria.getRequiredCategories().contains(profile.getCategory())) {
                return false;
            }
        }

        // 4. Occupation Check
        if (criteria.getAllowedOccupations() != null && !criteria.getAllowedOccupations().isEmpty()) {
            if (!criteria.getAllowedOccupations().contains(profile.getPrimaryOccupation())) {
                return false;
            }
        }

        // 5. Boolean Flag Checks
        if (criteria.getRequiresStudent() != null && criteria.getRequiresStudent() && !profile.isStudent()) {
            return false;
        }

        if (criteria.getRequiresFarmer() != null && criteria.getRequiresFarmer() && !profile.isFarmer()) {
            return false;
        }

        if (criteria.getRequiresDisability() != null && criteria.getRequiresDisability() && !profile.isHasDisability()) {
            return false;
        }

        // If we survived all checks, the citizen is eligible
        return true;
    }
}

package com.jansetu.backend.scheme.service;

import com.jansetu.backend.auth.entity.User;
import com.jansetu.backend.profile.entity.CitizenProfile;
import com.jansetu.backend.profile.service.CitizenProfileService;
import com.jansetu.backend.scheme.dto.RecommendationResponse;
import com.jansetu.backend.scheme.dto.SchemeResponse;
import com.jansetu.backend.scheme.entity.EligibilityCriteria;
import com.jansetu.backend.scheme.entity.Scheme;
import com.jansetu.backend.scheme.evaluator.EligibilityEvaluator;
import org.springframework.stereotype.Service;

import java.util.ArrayList;
import java.util.List;
import java.util.stream.Collectors;

@Service
public class MatchingService {

    private final CitizenProfileService profileService;
    private final SchemeService schemeService;
    private final EligibilityEvaluator evaluator;

    public MatchingService(CitizenProfileService profileService,
                           SchemeService schemeService,
                           EligibilityEvaluator evaluator) {
        this.profileService = profileService;
        this.schemeService = schemeService;
        this.evaluator = evaluator;
    }

    public List<RecommendationResponse> getRecommendations(User user) {
        // 1. Fetch user profile entity
        CitizenProfile profile = profileService.getProfileEntity(user);

        // 2. Fetch raw scheme entities
        List<Scheme> allSchemes = schemeService.getAllSchemesEntities();

        // 3. Filter using your evaluator, map to DTO, and wrap in RecommendationResponse
        return allSchemes.stream()
                .filter(scheme -> evaluator.isEligible(profile, scheme.getEligibilityCriteria()))
                .map(scheme -> {
                    SchemeResponse schemeDto = mapToResponse(scheme);

                    // ✅ FIX: Actually generate the dynamic reasons!
                    List<String> dynamicReasons = generateMatchReasons(profile, scheme.getEligibilityCriteria());

                    return RecommendationResponse.builder()
                            .scheme(schemeDto)
                            .matchPercentage(100)
                            .matchedCriteria(dynamicReasons) // ✅ FIX: Pass the dynamically generated list here!
                            .build();
                })
                .collect(Collectors.toList());
    }

    // --- NEW HELPER METHOD ---
    private List<String> generateMatchReasons(CitizenProfile profile, EligibilityCriteria criteria) {
        List<String> reasons = new ArrayList<>();

        // 1. Boolean Flags
        if (Boolean.TRUE.equals(criteria.getRequiresFarmer())) {
            reasons.add("You are a registered farmer.");
        }

        if (Boolean.TRUE.equals(criteria.getRequiresStudent())) {
            reasons.add("You are currently a student.");
        }

        if (Boolean.TRUE.equals(criteria.getRequiresDisability())) {
            reasons.add("You meet the specific disability criteria.");
        }

        // 2. Income limits
        if (criteria.getMaxFamilyIncome() != null) {
            reasons.add(String.format("Your annual family income is below ₹%,d.", criteria.getMaxFamilyIncome().longValue()));
        }

        // 3. Lists (Category and Occupation)
        if (criteria.getRequiredCategories() != null && !criteria.getRequiredCategories().isEmpty()) {
            // Since they passed the evaluator check, we know their specific category is in the list
            reasons.add("You belong to the eligible " + profile.getCategory() + " category.");
        }

        if (criteria.getAllowedOccupations() != null && !criteria.getAllowedOccupations().isEmpty()) {
            reasons.add("Your primary occupation (" + profile.getPrimaryOccupation() + ") is eligible.");
        }

        // 4. Age limits
        if (criteria.getMinAge() != null || criteria.getMaxAge() != null) {
            reasons.add("Your age falls within the required bracket for this scheme.");
        }

        // 5. Fallback
        if (reasons.isEmpty()) {
            reasons.add("You meet the baseline eligibility requirements for this scheme.");
        }

        return reasons;
    }

    // Helper method
    private SchemeResponse mapToResponse(Scheme scheme) {
        return new SchemeResponse(
                scheme.getId(),
                scheme.getName(),
                scheme.getDescription(),
                scheme.getEligibilityCriteria()
        );
    }
}

        scheme.setName(request.getName());
        scheme.setDescription(request.getDescription());

package com.jansetu.backend.auth.security;

import com.jansetu.backend.auth.service.JwtService;
import jakarta.servlet.FilterChain;
import jakarta.servlet.ServletException;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;
import org.springframework.lang.NonNull;
import org.springframework.security.authentication.UsernamePasswordAuthenticationToken;
import org.springframework.security.core.context.SecurityContextHolder;
import org.springframework.security.core.userdetails.UserDetails;
import org.springframework.security.core.userdetails.UserDetailsService;
import org.springframework.security.web.authentication.WebAuthenticationDetailsSource;
import org.springframework.stereotype.Component;
import org.springframework.web.filter.OncePerRequestFilter;

import java.io.IOException;

@Component
public class JwtAuthenticationFilter extends OncePerRequestFilter {

    private final JwtService jwtService;
    private final UserDetailsService userDetailsService;

    public JwtAuthenticationFilter(JwtService jwtService, UserDetailsService userDetailsService) {
        this.jwtService = jwtService;
        this.userDetailsService = userDetailsService;
    }

    @Override
    protected void doFilterInternal(
            @NonNull HttpServletRequest request,
            @NonNull HttpServletResponse response,
            @NonNull FilterChain filterChain
    ) throws ServletException, IOException {

        // 1. Look for the "Authorization" header
        final String authHeader = request.getHeader("Authorization");
        final String jwt;
        final String userEmail;

        // 2. If there is no token, or it doesn't start with "Bearer ", ignore it and move on
        if (authHeader == null || !authHeader.startsWith("Bearer ")) {
            filterChain.doFilter(request, response);
            return;
        }

        // 3. Extract the token (Remove "Bearer " from the string)
        jwt = authHeader.substring(7);
        userEmail = jwtService.extractUsername(jwt);

        // 4. If we found an email, and the user isn't already authenticated in this session...
        if (userEmail != null && SecurityContextHolder.getContext().getAuthentication() == null) {

            // Load the user from the database
            UserDetails userDetails = this.userDetailsService.loadUserByUsername(userEmail);

            // 5. If the token is valid, formally authenticate them with Spring Security
            if (jwtService.isTokenValid(jwt, userDetails.getUsername())) {
                UsernamePasswordAuthenticationToken authToken = new UsernamePasswordAuthenticationToken(
                        userDetails,
                        null,
                        userDetails.getAuthorities()
                );
                authToken.setDetails(new WebAuthenticationDetailsSource().buildDetails(request));

                // This is the VIP list update. We tell Spring: "This user is logged in!"
                SecurityContextHolder.getContext().setAuthentication(authToken);
            }
        }

        // 6. Continue to the next filter in the chain (or the controller)
        filterChain.doFilter(request, response);
    }
    @Override
    protected boolean shouldNotFilter(HttpServletRequest request) {
        String path = request.getServletPath();
        return path.startsWith("/api/v1/auth/") ||
                path.startsWith("/swagger-ui/") ||
                path.startsWith("/v3/api-docs");
    }
}

package com.jansetu.backend.auth.service;

import io.jsonwebtoken.Jwts;
import io.jsonwebtoken.security.Keys;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Service;

import javax.crypto.SecretKey;
import java.util.Date;

@Service
public class JwtService {

    // 1. Spring pulls these values from your environment variables/properties!
    @Value("${application.security.jwt.secret-key}")
    private String secretKey;

    @Value("${application.security.jwt.expiration}")
    private long jwtExpiration;

    // 2. Convert the secret string into a cryptographic key
    private SecretKey getSignInKey() {
        byte[] keyBytes = secretKey.getBytes();
        return Keys.hmacShaKeyFor(keyBytes);
    }

    // 3. Generate the actual VIP Badge
    public String generateToken(String email) {
        return Jwts.builder()
                .subject(email) // The "who" this badge belongs to
                .issuedAt(new Date(System.currentTimeMillis())) // The "when" it was printed
                .expiration(new Date(System.currentTimeMillis() + jwtExpiration)) // The "when" it expires
                .signWith(getSignInKey()) // The cryptographic signature
                .compact(); // Squishes it all into the final String token
    }
    // NEW: Extract the email (subject) from the token
    public String extractUsername(String token) {
        return Jwts.parser()
                .verifyWith(getSignInKey()) // Uses our secret key to check the signature!
                .build()
                .parseSignedClaims(token)
                .getPayload()
                .getSubject();
    }

    // NEW: Check if the token is valid for the specific user
    public boolean isTokenValid(String token, String userEmail) {
        final String extractedUsername = extractUsername(token);
        return (extractedUsername.equals(userEmail) && !isTokenExpired(token));
    }

    // NEW: Helper method to check expiration
    private boolean isTokenExpired(String token) {
        Date expiration = Jwts.parser()
                .verifyWith(getSignInKey())
                .build()
                .parseSignedClaims(token)
                .getPayload()
                .getExpiration();
        return expiration.before(new Date());
    }
}

package com.jansetu.backend.auth.service;

import com.jansetu.backend.auth.dto.LoginRequest;
import com.jansetu.backend.auth.dto.LoginResponse;
import com.jansetu.backend.auth.dto.RegisterRequest;
import com.jansetu.backend.auth.dto.RegisterResponse;
import com.jansetu.backend.auth.entity.Role;
import com.jansetu.backend.auth.entity.User;
import com.jansetu.backend.auth.repository.UserRepository;
import lombok.extern.slf4j.Slf4j;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.security.crypto.password.PasswordEncoder;
import org.springframework.stereotype.Service;
import com.jansetu.backend.auth.exception.EmailAlreadyExistsException;
import com.jansetu.backend.auth.exception.InvalidCredentialsException;
import java.util.Optional;

@Slf4j
@Service
public class AuthService {

    private final UserRepository userRepository;
    private final PasswordEncoder passwordEncoder;
    private final JwtService jwtService;

    @Value("${application.security.jwt.expiration}")
    private long jwtExpiration;

    public AuthService(UserRepository userRepository, PasswordEncoder passwordEncoder, JwtService jwtService) {
        this.userRepository = userRepository;
        this.passwordEncoder = passwordEncoder;
        this.jwtService = jwtService;
    }

    public RegisterResponse register(RegisterRequest request) {
        log.info("Attempting to register new user with email: {}", request.email());

        Optional<User> existingUser = userRepository.findByEmail(request.email());
        if (existingUser.isPresent()) {
            log.warn("Registration failed - email already exists: {}", request.email());
            throw new EmailAlreadyExistsException(); // Note: You can swap this to a custom BusinessException later!
        }

        String hashedPassword = passwordEncoder.encode(request.password());

        User user = new User();
        user.setEmail(request.email());
        user.setPasswordHash(hashedPassword);
        user.setRole(Role.CITIZEN);
        user.setEnabled(true);

        User savedUser = userRepository.save(user);

        log.info("Successfully registered user: {}", savedUser.getEmail());
        return new RegisterResponse(savedUser.getId(), "User registered successfully");
    }

    public LoginResponse login(LoginRequest request) {
        log.info("Login attempt for email: {}", request.email());

        // 1. Find the user by email
        User user = userRepository.findByEmail(request.email())
                .orElseThrow(() -> {
                    log.warn("Login failed - user not found: {}", request.email());
                    return new InvalidCredentialsException();
                });

        // 2. Check if the password matches the hash in the database
        if (!passwordEncoder.matches(request.password(), user.getPasswordHash())) {
            log.warn("Login failed - bad password for email: {}", request.email());
            throw new InvalidCredentialsException();
        }

        // 3. Generate the JWT
        String jwtToken = jwtService.generateToken(user.getEmail());

        log.info("User {} successfully logged in", user.getEmail());
        // 4. Return the VIP Badge
        return new LoginResponse(jwtToken, "Bearer", jwtExpiration);
    }
}

import java.util.stream.Collectors;

@Slf4j
@RestControllerAdvice
public class GlobalExceptionHandler {
    @ExceptionHandler(InvalidCredentialsException.class)
    public ResponseEntity<ErrorResponse> handleInvalidCredentials(
            InvalidCredentialsException ex) {

        return buildResponse(
                HttpStatus.UNAUTHORIZED,
                "Unauthorized",
                ex.getMessage()
        );
    }

    @ExceptionHandler(EmailAlreadyExistsException.class)
    public ResponseEntity<ErrorResponse> handleEmailAlreadyExists(
            EmailAlreadyExistsException ex) {

        return buildResponse(
                HttpStatus.CONFLICT,
                "Conflict",
                ex.getMessage()
        );
    }

    // Handle Not Found exceptions (404)
    @ExceptionHandler({ProfileNotFoundException.class, SchemeNotFoundException.class})
    public ResponseEntity<ErrorResponse> handleNotFound(BusinessException ex) {
        return buildResponse(HttpStatus.NOT_FOUND, "Resource Not Found", ex.getMessage());
    }

    // Handle Conflict exceptions (409)
    @ExceptionHandler(DuplicateApplicationException.class)
    public ResponseEntity<ErrorResponse> handleConflict(DuplicateApplicationException ex) {
        return buildResponse(HttpStatus.CONFLICT, "Conflict", ex.getMessage());
    }

    // Handle Bad Request exceptions (400)
    @ExceptionHandler(IneligibleException.class)
    public ResponseEntity<ErrorResponse> handleBadRequest(IneligibleException ex) {
        return buildResponse(HttpStatus.BAD_REQUEST, "Bad Request", ex.getMessage());
    }

    // Handle Validation Errors (400)
    @ExceptionHandler(MethodArgumentNotValidException.class)
    public ResponseEntity<ErrorResponse> handleValidationExceptions(MethodArgumentNotValidException ex) {
        // Collect all validation error messages into a single comma-separated string
        String errorMessage = ex.getBindingResult().getFieldErrors().stream()
                .map(error -> error.getField() + ": " + error.getDefaultMessage())
                .collect(Collectors.joining(", "));

        return buildResponse(HttpStatus.BAD_REQUEST, "Validation Failed", errorMessage);
    }

    // Fallback for unexpected exceptions (500)
    @ExceptionHandler(Exception.class)
    public ResponseEntity<ErrorResponse> handleGenericException(Exception ex) {
        // This prints the error in your console so you can actually debug it
        log.error("An unexpected error occurred: ", ex);
        return buildResponse(HttpStatus.INTERNAL_SERVER_ERROR, "Internal Server Error", "An unexpected error occurred. Please try again later.");
    }

    // Helper method to keep code clean
    private ResponseEntity<ErrorResponse> buildResponse(HttpStatus status, String error, String message) {
        ErrorResponse response = ErrorResponse.builder()
                .timestamp(LocalDateTime.now())
                .status(status.value())
                .error(error)
                .message(message)
                .build();
        return new ResponseEntity<>(response, status);
    }

package com.jansetu.backend.profile.service;

import com.jansetu.backend.auth.entity.User;
import com.jansetu.backend.profile.dto.ProfileRequest;
import com.jansetu.backend.profile.dto.ProfileResponse;
import com.jansetu.backend.profile.entity.CitizenProfile;
import com.jansetu.backend.profile.exception.ProfileNotFoundException;
import com.jansetu.backend.profile.repository.CitizenProfileRepository;
import org.springframework.stereotype.Service;

@Service
public class CitizenProfileService {

    private final CitizenProfileRepository profileRepository;

    public CitizenProfileService(CitizenProfileRepository profileRepository) {
        this.profileRepository = profileRepository;
    }

    // --- 1. CREATE PROFILE ---
    public ProfileResponse createProfile(User user, ProfileRequest request) {
        // Business Rule: A user can only have ONE profile.
        if (profileRepository.findByUser(user).isPresent()) {
            throw new IllegalStateException("A profile already exists for this user.");
        }

        CitizenProfile profile = new CitizenProfile();
        profile.setUser(user); // Establish the OneToOne link

        updateProfileFields(profile, request);

        CitizenProfile savedProfile = profileRepository.save(profile);
        return mapToResponse(savedProfile);
    }

    // --- 2. GET PROFILE ---
    public ProfileResponse getProfile(User user) {
        return mapToResponse(getProfileEntity(user));
    }

    // --- 3. UPDATE PROFILE ---
    public ProfileResponse updateProfile(User user, ProfileRequest request) {
        CitizenProfile profile = profileRepository.findByUser(user)
                .orElseThrow(() -> new ProfileNotFoundException("Profile not found"));

        updateProfileFields(profile, request);

        CitizenProfile updatedProfile = profileRepository.save(profile);
        return mapToResponse(updatedProfile);
    }

    // --- HELPER: Map Data from DTO to Entity ---
    private void updateProfileFields(CitizenProfile profile, ProfileRequest request) {
        profile.setFirstName(request.getFirstName());
        profile.setLastName(request.getLastName());
        profile.setDateOfBirth(request.getDateOfBirth());
        profile.setGender(request.getGender());
        profile.setState(request.getState());
        profile.setDistrict(request.getDistrict());
        profile.setPincode(request.getPincode());
        profile.setCategory(request.getCategory());
        profile.setAnnualFamilyIncome(request.getAnnualFamilyIncome());
        profile.setMaritalStatus(request.getMaritalStatus());
        profile.setPrimaryOccupation(request.getPrimaryOccupation());

        // Note: Boolean objects in DTO use getter, boolean primitives in Entity use setter
        profile.setStudent(request.getIsStudent());
        profile.setFarmer(request.getIsFarmer());
        profile.setHasDisability(request.getHasDisability());
    }
    public CitizenProfile getProfileByEmail(String email) {
        return profileRepository.findByUserEmail(email)
                .orElseThrow(() -> new ProfileNotFoundException("Profile not found"));
    }
    public CitizenProfile getProfileEntity(User user) {
        // Use the ID instead of the User object reference
        return profileRepository.findByUser_Id(user.getId())
                .orElseThrow(() -> new ProfileNotFoundException("Profile not found"));
    }

    // --- HELPER: Map Data from Entity to DTO ---
    private ProfileResponse mapToResponse(CitizenProfile profile) {
        return ProfileResponse.builder()
                .profileId(profile.getId())
                .firstName(profile.getFirstName())
                .lastName(profile.getLastName())
                .dateOfBirth(profile.getDateOfBirth())
                .gender(profile.getGender())
                .state(profile.getState())
                .district(profile.getDistrict())
                .pincode(profile.getPincode())
                .category(profile.getCategory())
                .annualFamilyIncome(profile.getAnnualFamilyIncome())
                .maritalStatus(profile.getMaritalStatus())
                .primaryOccupation(profile.getPrimaryOccupation())
                .isStudent(profile.isStudent())
                .isFarmer(profile.isFarmer())
                .hasDisability(profile.isHasDisability())
                .lastUpdated(profile.getUpdatedAt()) // Returns the auto-generated timestamp
                .build();
    }
}

package com.jansetu.backend.ai.service;

import com.jansetu.backend.ai.context.ContextBuilder;
import com.jansetu.backend.ai.dto.ChatRequest;
import com.jansetu.backend.ai.dto.ExplainRequest;
import com.jansetu.backend.ai.prompt.PromptBuilder;
import org.springframework.ai.chat.client.ChatClient;
import org.springframework.stereotype.Service;

@Service
public class AIService {

    private final ChatClient chatClient;

    private final PromptBuilder promptBuilder;

    private final ContextBuilder contextBuilder;

    public AIService(
            ChatClient.Builder builder,
            PromptBuilder promptBuilder,
            ContextBuilder contextBuilder
    ) {
        this.chatClient = builder.build();
        this.promptBuilder = promptBuilder;
        this.contextBuilder = contextBuilder;
    }

    public String explainEligibility(ExplainRequest request) {

        String systemPrompt = """
                You are an expert advisor for JanSetu.

                Explain eligibility.

                Keep it under four sentences.

                Use bullets.

                Do not hallucinate.

                Only use supplied information.
                """;

        String userPrompt = String.format("""
                        User Profile:
                        %s

                        Scheme:
                        %s

                        Description:
                        %s

                        Matched Criteria:
                        %s
                        """,
                request.userProfileSummary(),
                request.schemeName(),
                request.schemeDescription(),
                String.join(", ", request.matchedCriteria())
        );

        return chatClient.prompt()
                .system(systemPrompt)
                .user(userPrompt)
                .call()
                .content();
    }

    public String chat(ChatRequest request) {

        String prompt = promptBuilder.buildChatPrompt(
                contextBuilder.buildProfileSummary(),
                contextBuilder.buildRecommendationsSummary(),
                contextBuilder.buildApplicationSummary(),
                request.message()
        );

        return chatClient.prompt()
                .user(prompt)
                .call()
                .content();
    }

}

package com.jansetu.backend.application.service;

import com.jansetu.backend.application.dto.ApplicationRequest;
import com.jansetu.backend.application.dto.ApplicationResponse;
import com.jansetu.backend.application.entity.Application;
import com.jansetu.backend.application.enums.ApplicationStatus;
import com.jansetu.backend.application.exception.DuplicateApplicationException;
import com.jansetu.backend.application.exception.IneligibleException;
import com.jansetu.backend.application.repository.ApplicationRepository;
import com.jansetu.backend.auth.entity.User;
import com.jansetu.backend.profile.entity.CitizenProfile;
import com.jansetu.backend.profile.service.CitizenProfileService;
import com.jansetu.backend.scheme.entity.Scheme;
import com.jansetu.backend.scheme.evaluator.EligibilityEvaluator;
import com.jansetu.backend.scheme.exception.SchemeNotFoundException;
import com.jansetu.backend.scheme.repository.SchemeRepository;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.util.List;

@Service
public class ApplicationService {

    private final ApplicationRepository applicationRepository;
    private final CitizenProfileService profileService;
    private final SchemeRepository schemeRepository;
    private final EligibilityEvaluator evaluator;

    public ApplicationService(ApplicationRepository applicationRepository,
                              CitizenProfileService profileService,
                              SchemeRepository schemeRepository,
                              EligibilityEvaluator evaluator) {
        this.applicationRepository = applicationRepository;
        this.profileService = profileService;
        this.schemeRepository = schemeRepository;
        this.evaluator = evaluator;
    }

    @Transactional
    public ApplicationResponse submitApplication(User user, ApplicationRequest request) {
        // 1. Fetch Profile and Scheme
        CitizenProfile profile = profileService.getProfileEntity(user);
        Scheme scheme = schemeRepository.findById(request.getSchemeId())
                .orElseThrow(() -> new SchemeNotFoundException("Scheme not found"));

        // 2. Validate Eligibility
        if (!evaluator.isEligible(profile, scheme.getEligibilityCriteria())) {
            throw new IneligibleException("User is not eligible");
        }

        // 3. Validate Duplicates (Exclude Rejected/Withdrawn)
        if (applicationRepository.existsByCitizenProfileAndSchemeAndStatusNot(
                profile, scheme, ApplicationStatus.REJECTED)) { // Simplification: adjust logic as needed
            throw new DuplicateApplicationException(
                    "An active application already exists for this scheme"
            );        }

        // 4. Create and Save
        Application application = Application.builder()
                .citizenProfile(profile)
                .scheme(scheme)
                .status(ApplicationStatus.SUBMITTED)
                .build();

        Application savedApp = applicationRepository.save(application);

        return mapToResponse(savedApp);
    }
    public List<ApplicationResponse> getMyApplications(User user) {
        CitizenProfile profile = profileService.getProfileEntity(user);

        return applicationRepository.findByCitizenProfile(profile).stream()
                .map(this::mapToResponse)
                .toList();
    }

    private ApplicationResponse mapToResponse(Application app) {
        return ApplicationResponse.builder()
                .applicationId(app.getId())
                .schemeName(app.getScheme().getName())
                .schemeId(app.getScheme().getId())
                .status(app.getStatus())
                .remarks(app.getRemarks())
                .submittedAt(app.getSubmittedAt())
                .updatedAt(app.getUpdatedAt())
                .build();
    }
}
